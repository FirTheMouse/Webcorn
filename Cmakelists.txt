cmake_minimum_required(VERSION 3.15)
project(Webcorn)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2")

find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
    set(CMAKE_CXX_COMPILER_LAUNCHER ${CCACHE_PROGRAM})
endif()

set(MBEDTLS_PSA_CRYPTO_C OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory(ext/mbedtls)

add_executable(webcorn main.cpp)
add_custom_command(TARGET webcorn POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:webcorn> ${CMAKE_SOURCE_DIR}/webcorn
)

option(USE_TLS "Enable TLS" OFF)
option(USE_ASAN "Enable AddressSanitizer" OFF)
option(USE_TSAN "Enable ThreadSanitizer" OFF)

if(USE_TLS)
    target_compile_definitions(webcorn PRIVATE USE_TLS=1)
else()
    target_compile_definitions(webcorn PRIVATE USE_TLS=0)
endif()

if(USE_ASAN)
    target_compile_options(webcorn PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
    target_link_options(webcorn PRIVATE -fsanitize=address)
endif()

if(USE_TSAN)
    target_compile_options(webcorn PRIVATE -fsanitize=thread -fno-omit-frame-pointer -g)
    target_link_options(webcorn PRIVATE -fsanitize=thread)
endif()

target_include_directories(webcorn PRIVATE
    GDSL
    ext/mbedtls/include
    ext/mbedtls/tf-psa-crypto/include
)

target_link_libraries(webcorn PRIVATE
    mbedtls
)
